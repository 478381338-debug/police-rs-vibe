def _tile_usable(data: bytes, ctype: str) -> bool:
    """上游会把空瓦片当成功返回，这类瓦片一旦缓存就会在底图上留下白洞。"""
    if not data or len(data) < 512 or not ctype.startswith("image/"):
        return False
    try:
        from PIL import Image
        import io as _io
        im = Image.open(_io.BytesIO(data))
        im.verify()
        im = Image.open(_io.BytesIO(data)).convert("RGB").resize((16, 16))
    except Exception:
        return False
    # 纯白瓦片＝上游没有该处数据（例如海面 or 超出覆盖范围），按失败处理。
    return not all(mx >= 250 and mx - mn < 2 for mn, mx in im.getextrema())
